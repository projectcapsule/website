{{- $name := .Get "src" -}} {{- $source := .Page.Parent.Resources.Get $name -}} {{- if not $source -}} {{- errorf "linked-yaml: %s not found beside %s" $name .Page.File.Path -}} {{- else -}} {{- $links := dict -}} {{- range transform.Unmarshal .Inner -}} {{- if not (hasPrefix .url "/docs/") -}} {{- errorf "linked-yaml: expected a documentation link, got %q" .url -}} {{- end -}} {{- $comment := printf "# %s" .comment -}} {{- if index $links $comment -}} {{- errorf "linked-yaml: duplicate annotation %q" .comment -}} {{- end -}} {{- $links = merge $links (dict $comment .) -}} {{- end -}} {{/* A linked comment introduces a YAML block. Its first property and all deeper-indented lines belong to the link; the next sibling does not. */}} {{- $segments := slice -}} {{- $lines := slice -}} {{- $active := dict -}} {{- $indent := 0 -}} {{- $hasContent := false -}} {{- $seen := slice -}} {{- range split (strings.TrimSuffix "\n" $source.Content) "\n" -}} {{- $text := trim . " \t\r\n" -}} {{- $depth := sub (len .) (len (replaceRE "^ +" "" .)) -}} {{- if and $active $hasContent $text (le $depth $indent) -}} {{- $segments = $segments | append (dict "lines" $lines "link" $active) -}} {{- $lines = slice -}} {{- $active = dict -}} {{- end -}} {{- with index $links $text -}} {{- if or $active (in $seen $text) -}} {{- errorf "linked-yaml: nested or repeated annotation %q in %s" $text $name -}} {{- end -}} {{- if $lines -}} {{- $segments = $segments | append (dict "lines" $lines "link" (dict)) -}} {{- $lines = slice -}} {{- end -}} {{- $active = . -}} {{- $indent = $depth -}} {{- $hasContent = false -}} {{- $seen = $seen | append $text -}} {{- end -}} {{- $lines = $lines | append . -}} {{- if and $active $text (not (hasPrefix $text "#")) -}} {{- $hasContent = true -}} {{- end -}} {{- end -}} {{- $segments = $segments | append (dict "lines" $lines "link" $active) -}} {{- if ne (len $seen) (len $links) -}} {{- errorf "linked-yaml: an annotated comment is missing from %s" $name -}} {{- end -}} {{- $code := "" -}} {{- range $segments -}} {{- $chunk := printf "%s\n" (delimit .lines "\n") -}} {{/* Inline highlighting keeps the complete YAML inside one copyable code element. Remove only its outer code tag; retain all highlighted text. */}} {{- $html := highlight $chunk "yaml" "hl_inline=true" | replaceRE `^]*>|$` "" -}} {{/* Older Chroma versions can add whitespace to folded YAML scalars. Preserve the source exactly if highlighting changes its text. */}} {{- if ne (htmlUnescape (replaceRE `<[^>]*>` "" $html)) $chunk -}} {{- $html = htmlEscape $chunk -}} {{- end -}} {{- with .link -}} {{- $label := printf "Read documentation: %s" .comment -}} {{- $code = printf `%s%s` $code (.url | relURL | htmlEscape) ($label | htmlEscape) $html -}} {{- else -}} {{- $code = printf `%s%s` $code $html -}} {{- end -}} {{- end -}}
